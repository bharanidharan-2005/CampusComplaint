class ComplaintDetailView(APIView):
    """Full detail of a single complaint including the student's identity info."""
    permission_classes = [IsAuthenticated]

    def get(self, request, pk):
        try:
            complaint = Complaint.objects.select_related('student', 'student__student_profile').get(pk=pk)
        except Complaint.DoesNotExist:
            return Response({"error": "Complaint not found."}, status=status.HTTP_404_NOT_FOUND)

        student = complaint.student
        profile = getattr(student, 'student_profile', None)

        priority_map = {
            'Escalated': 'Critical',
            'Pending': 'High',
            'In Progress': 'Medium',
            'Completed': 'Low',
            'Resolved': 'Low',
        }

        image_url = None
        if complaint.image:
            image_url = request.build_absolute_uri(complaint.image.url)

        return Response({
            "id": complaint.id,
            "title": complaint.title,
            "description": complaint.description,
            "category": complaint.category,
            "status": complaint.status,
            "priority": priority_map.get(complaint.status, 'Medium'),
            "created_at": complaint.created_at.strftime("%Y-%m-%d %H:%M") if complaint.created_at else "",
            "location": complaint.location,
            "person_name": complaint.person_name,
            "image": image_url,
            "student": {
                "name": getattr(profile, 'full_name', None) or complaint.student_name or getattr(student, 'email', ''),
                "register_number": getattr(profile, 'register_number', None) or complaint.register_number or '',
                "department": getattr(getattr(profile, 'department', None), 'department_name', None) or complaint.department or '',
                "year": getattr(profile, 'study_year', None) or complaint.year or '',
                "email": getattr(student, 'email', '')
            }
        }, status=status.HTTP_200_OK)
